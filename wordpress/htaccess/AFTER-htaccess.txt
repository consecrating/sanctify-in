# BEGIN Sanctify File Protection
# Added 2026-09-08. Blocks backup archives, database dumps and config disclosure.
#
# -- GOOGLEBOT SAFETY  deliberate design constraints ------------------------
#  * Every rule below matches FILE NAMES AND EXTENSIONS ONLY. There is no
#    User-Agent matching, no IP or country blocking and no rate limiting
#    anywhere in this block, so no crawler can ever be caught by it.
#  * Nothing Googlebot needs in order to render or index the site is touched:
#    CSS, JS, images, fonts, /wp-content/uploads/, robots.txt, sitemap.xml and
#    ads.txt all remain fully accessible.
#  * No X-Robots-Tag and no noindex is set anywhere in this block.
#  * Googlebot has no reason to fetch a .zip backup or a debug log, so denying
#    those cannot affect crawling, indexing or rankings.
#  * Verified before deployment: no page on the site links to an archive, and
#    there are no archives in /wp-content/uploads/, so nothing user-facing
#    breaks.
# ---------------------------------------------------------------------------

# 1. Archives, database dumps, and editor/upgrade leftovers.
<FilesMatch "\.(zip|tar|gz|tgz|bz2|xz|7z|rar|sql|dump|bak|backup|old|save|orig|swp|swo|tmp|log)$">
    Require all denied
</FilesMatch>

# 2. Re-allow sitemaps in case an SEO plugin ever emits a gzipped one.
#    Deliberately placed AFTER rule 1: Apache 2.4 merges matching sections and
#    the later directive wins, so this is what keeps a .xml.gz sitemap crawlable.
<FilesMatch "^sitemap[^/]*\.xml(\.gz)?$">
    Require all granted
</FilesMatch>

# 3. Apache config files. This host does NOT apply the usual "^\.ht" deny, so
#    .htaccessold and .htaccess.phpupgrader.* were being served in full.
#    Note: this blocks HTTP access only  Apache still reads .htaccess normally,
#    so WP Rocket's own cache .htaccess keeps working.
<FilesMatch "^\.ht">
    Require all denied
</FilesMatch>

# 4. Version disclosure and sensitive dotfiles.
<FilesMatch "^(readme\.html|license\.txt|wp-config-sample\.php|\.user\.ini|\.ftpquota|error_log)$">
    Require all denied
</FilesMatch>

# 5. wp-config in any backup guise (wp-configbackup*.php.zip, wp-config.php.bak, ...)
<FilesMatch "^wp-config.*\.(php\.zip|zip|bak|backup|old|save|orig|txt|json|_bak)$">
    Require all denied
</FilesMatch>

# 6. No directory listings. Confirmed permitted on this host by an isolated
#    probe before deploying, because a rejected Options directive would 500 the
#    entire site.
Options -Indexes

# END Sanctify File Protection
